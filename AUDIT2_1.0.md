#if API
using System; using System.Collections.Generic; using System.Linq;
using System.Reflection; using System.Reflection.Emit; using HarmonyLib;
namespace Auga {
  public static partial class API {
    private static readonly Assembly _targetAssembly;
    static API() {
      _targetAssembly = LoadAssembly();
      if (_targetAssembly == null) return;
      var harmony = new Harmony("mods.randyknapp.auga.API");
      foreach (var m in typeof(API).GetMethods(BindingFlags.DeclaredOnly | BindingFlags.Static | BindingFlags.Public)
                                   .Where(m => m.Name != nameof(LoadAssembly)))
        harmony.Patch(m, transpiler: new HarmonyMethod(AccessTools.DeclaredMethod(typeof(API), nameof(Transpiler))));
    }
    private static IEnumerable<CodeInstruction> Transpiler(IEnumerable<CodeInstruction> _, MethodBase original) {
      var parameters = original.GetParameters().Select(p => {
        var t = p.ParameterType;
        if (t.Assembly == Assembly.GetExecutingAssembly()) t = _targetAssembly.GetType(t.FullName ?? string.Empty) ?? t;
        return t; }).ToArray();
      var target = _targetAssembly.GetType("Auga.API").GetMethod(original.Name, parameters);
      for (var i = 0; i < parameters.Length; ++i) yield return CodeInstruction.LoadArgument(i);
      yield return new CodeInstruction(OpCodes.Call, target);
      yield return new CodeInstruction(OpCodes.Ret);
    }
    public static Assembly LoadAssembly() =>
      AppDomain.CurrentDomain.GetAssemblies().FirstOrDefault(a => a.GetName().Name == "Auga");
  }
}
#endif

Two deliberate differences from the 2021 original, both required: (a) `IsLoaded` is NOT excluded from patching any more — the original excluded it because its own body was `LoadAssembly() != null`, whereas API.cs:42 under `API` returns literal `false`; leaving the old exclusion in place would make every consumer's `if (Auga.API.IsLoaded())` permanently false and kill the whole API. Patched, it forwards to the real `IsLoaded()` (returns true); unpatched (Auga absent) it correctly returns false. (b) `CodeInstruction.LoadArgument(i)` instead of `new CodeInstruction(OpCodes.Ldarg, i)` (API.External.cs:48) — the boxed int32 operand on an InlineVar opcode is an operand-size ambiguity in the emitter; LoadArgument picks the right ldarg form. Keep `Harmony("mods.randyknapp.auga.API")` distinct from Auga's own id `randyknapp.mods.auga` (Auga.cs:89).
